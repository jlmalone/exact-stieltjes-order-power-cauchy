\title{Exact Stieltjes Order of Power-Cauchy Minors\\
via Fixed-Trace Unitary Orbits}
\author{}
\date{}

\begin{document}
\maketitle

\begin{abstract}
For positive $a$ and two strictly ordered real spectra, consider the
determinant with entries $(s+t_i+x_j)^{-a}$. We prove that, as a function of
the common shift $s$, this minor is a generalized Stieltjes transform of exact
order $n(a+n-1)$. The representing probability law is explicit: it is the sum
of two independent unitary-orbital linear statistics conditioned on a common
fixed-trace Laguerre spectrum. We determine its full support, mean, and
variance. The representation yields sharp support-and-mean determinant bounds,
a quantitative variance refinement, and a strict Hausdorff-moment hierarchy
for the normalized expansion at infinity. Andr\'eief's identity extends the
result to determinants of generalized Markov transforms generated by strict
Chebyshev systems and to their confluent Wronskians. The proof combines the
Gamma-Laplace representation, the Harish-Chandra--Itzykson--Zuber integral,
and the Laguerre--Selberg normalization. Independent double-alternant
asymptotics confirm the exponent and normalization.
\end{abstract}

\noindent\textbf{Keywords.}
Generalized Stieltjes transform; exact Stieltjes order; Cauchy kernel; total
positivity; HCIZ integral; fixed-trace Laguerre ensemble; Chebyshev system;
Wronskian.

\section{Introduction}

For $a>0$, strictly increasing real vectors $t=(t_1,\ldots,t_n)$ and
$x=(x_1,\ldots,x_n)$, and $s>0$, consider the power-Cauchy minor
\begin{equation}\label{eq:kernel-minor}
K_{a,n}(s;t,x)
=\det\left[(s+t_i+x_j)^{-a}\right]_{i,j=1}^{n}.
\end{equation}
We assume $t_1+x_1\ge0$, so every entry is defined for $s>0$ and every
representing measure below is supported on the nonnegative half-line.

The positivity and sign regularity of generalized Cauchy kernels belong to
classical total-positivity theory; see, for example,
\citet{DetteMunk1996}. Determinant integral representations for analytic
product kernels and their relation to spherical series and functions of matrix
argument were developed by \citet{GrossRichards1989}. Determinants of moment
and Laplace-transform matrices also occur in Lindsay-transform theory
\citep{KokonendjiSeshadri1996}, while \citet{DimitrovXu2019} proved complete
monotonicity of Wronskians of Laplace transforms.

Our question concerns a stronger property in the common shift. For which
$\alpha$ does \eqref{eq:kernel-minor} admit a positive representation as a
mixture of $(s+y)^{-\alpha}$, and what is the smallest such $\alpha$? Exact
generalized Stieltjes order was developed systematically by
\citet{KarpPrilepkina2012}; related real-variable characterizations appear in
\citet{Sokal2010,KoumandosPedersen2018}, and product representations were
studied by \citet{GomilkoTomilov2024}.

We obtain the exact order
\[
n(a+n-1)=na+n(n-1).
\]
The linear term $na$ is the order predicted by multiplying $n$ scalar kernels.
The additional quadratic term records the cancellation forced by alternating
in both spectra. The representing law is a compactly supported random-matrix
object: the sum of two independent unitary-orbital statistics sharing one
fixed-trace Laguerre spectrum. This representation yields exact support and
moment information that is not visible from determinant positivity alone.

The main consequences are:
\begin{enumerate}[leftmargin=2em]
  \item the exact order and an explicit probability representation;
  \item the full support, mean, and variance of the representing variable;
  \item two-sided determinant bounds and a variance-refined lower bound;
  \item a strict Hausdorff-moment hierarchy for the expansion at infinity;
  \item the same exact order for Chebyshev-Markov determinants and their
  confluent Wronskians.
\end{enumerate}

\section{Generalized Stieltjes order}

\begin{definition}
For $\alpha>0$, let $\cS_\alpha$ be the class of functions
\begin{equation}\label{eq:stieltjes-class}
f(s)=c+\int_{[0,\infty)}(s+y)^{-\alpha}\,\nu(dy),\qquad s>0,
\end{equation}
where $c\ge0$ and $\nu$ is a positive measure making the integral finite. For
a nonzero function in $\bigcup_{\alpha>0}\cS_\alpha$, its exact Stieltjes
order is
\[
\alpha_*(f)=\inf\{\alpha>0:f\in\cS_\alpha\}.
\]
\end{definition}

Write
\begin{equation}\label{eq:vandermonde}
\Delta(z)=\prod_{1\le i<j\le n}(z_j-z_i).
\end{equation}
Our convention makes $\Delta(z)>0$ for an increasing vector.

\section{Fixed-trace orbital representation}

Put
\begin{equation}\label{eq:lambda-c}
\lambda=n(a+n-1),\qquad
c_{a,n}=\prod_{k=0}^{n-1}\frac{(a)_k}{k!}.
\end{equation}
Let $V=\operatorname{diag}(v_1,\ldots,v_n)$, where $v$ has density
\begin{equation}\label{eq:fixed-trace-density}
\frac{1}{Z_{a,n}}\Delta(v)^2\prod_{i=1}^n v_i^{a-1}
\end{equation}
on the simplex $v_i>0$, $\sum_i v_i=1$, with
\begin{equation}\label{eq:Z}
Z_{a,n}
=\frac{\prod_{j=1}^{n}\Gamma(j+1)\Gamma(a+j-1)}
       {\Gamma(\lambda)}.
\end{equation}
This is the unitary fixed-trace Laguerre eigenvalue density. Such laws are
standard in the theory of random density matrices
\citep{SommersZyczkowski2004}.

Let $U,W$ be independent normalized-Haar $U(n)$ matrices, independent of
$V$, and define
\begin{equation}\label{eq:Y}
T=\operatorname{diag}(t_1,\ldots,t_n),\quad
X=\operatorname{diag}(x_1,\ldots,x_n),
\qquad
Y=\tr(TUVU^*)+\tr(XWVW^*).
\end{equation}

\begin{theorem}[Orbital representation and exact order]
\label{thm:main}
For $a>0$ and strictly increasing real vectors $t,x$ satisfying
$t_1+x_1\ge0$,
\begin{equation}\label{eq:main-representation}
K_{a,n}(s;t,x)
=c_{a,n}\Delta(t)\Delta(x)\E(s+Y)^{-\lambda}.
\end{equation}
The support of $Y$ is
\begin{equation}\label{eq:support}
\supp Y=[t_1+x_1,t_n+x_n],
\end{equation}
where the interval is a singleton when $n=1$. In particular,
\begin{equation}\label{eq:exact-order}
\alpha_*(K_{a,n})=n(a+n-1).
\end{equation}
\end{theorem}

\begin{proof}
The Gamma-Laplace formula gives
\[
(s+t_i+x_j)^{-a}
=\frac{1}{\Gamma(a)}\int_0^\infty
e^{-(s+t_i+x_j)u}u^{a-1}\,du.
\]
Andr\'eief's identity therefore yields
\begin{equation}\label{eq:andreief}
K_{a,n}
=\frac{1}{n!\Gamma(a)^n}
\int_{(0,\infty)^n}
\det[e^{-t_i u_k}]\det[e^{-x_j u_k}]
e^{-s\sum_k u_k}\prod_k u_k^{a-1}\,du.
\end{equation}

Set $r=\sum_k u_k$ and $u_k=rv_k$. The simplex Jacobian is $r^{n-1}$.
Let
\[
N=\frac{n(n-1)}2,\qquad C_n=\prod_{j=1}^{n-1}j!.
\]
The unitary HCIZ formula, in the normalization reviewed by
\citet{McSwiggen2020}, gives
\begin{equation}\label{eq:hciz-det}
\det[e^{-rt_i v_j}]
=\frac{(-r)^N\Delta(t)\Delta(v)}{C_n}
\int_{U(n)}e^{-r\tr(TUVU^*)}\,dU.
\end{equation}
Use \eqref{eq:hciz-det} for both determinants in
\eqref{eq:andreief}. The signs cancel, and the radial power is
\[
n(a-1)+(n-1)+2N
=n(a+n-1)-1
=\lambda-1.
\]
Thus
\begin{align}
K_{a,n}
&=\frac{\Delta(t)\Delta(x)}{n!\Gamma(a)^nC_n^2}
\int_{\sum v_i=1}\Delta(v)^2\prod_i v_i^{a-1}\nonumber\\
&\qquad\times
\E_{U,W}\int_0^\infty r^{\lambda-1}e^{-r(s+Y)}\,dr\,dv.
\label{eq:radialized}
\end{align}
The radial integral equals $\Gamma(\lambda)(s+Y)^{-\lambda}$.

For completeness, the Laguerre normalization is
\begin{equation}\label{eq:laguerre-norm}
\int_{(0,\infty)^n}e^{-\sum_i u_i}\Delta(u)^2
\prod_i u_i^{a-1}\,du
=n!\prod_{k=0}^{n-1}k!\Gamma(a+k).
\end{equation}
Indeed, Andr\'eief applied to the monic Laguerre polynomials reduces the
integral to $n!$ times the product of their squared norms
$k!\Gamma(a+k)$. A radial decomposition of
\eqref{eq:laguerre-norm} gives \eqref{eq:Z}. Direct cancellation now yields
\[
\frac{\Gamma(\lambda)Z_{a,n}}
{n!\Gamma(a)^nC_n^2}
=\prod_{k=0}^{n-1}\frac{(a)_k}{k!},
\]
proving \eqref{eq:main-representation}.

Every positive semidefinite trace-one matrix $Q$ satisfies
\[
t_1\le\tr(TQ)\le t_n.
\]
The analogous inequality for $X$ proves containment in the interval
\eqref{eq:support}. The closure of the simplex support contains the rank-one
spectrum $(1,0,\ldots,0)$. Rank-one unitary orbits give every Rayleigh
quotient in $[t_1,t_n]$, and independently every quotient in $[x_1,x_n]$.
Every point of their sum interval is therefore in the support.

The representation places $K_{a,n}$ in $\cS_\lambda$, while dominated
convergence gives
\begin{equation}\label{eq:leading-asymptotic}
s^\lambda K_{a,n}(s;t,x)
\longrightarrow c_{a,n}\Delta(t)\Delta(x)>0.
\end{equation}
If $K_{a,n}\in\cS_\beta$ for $0<\beta<\lambda$, its decay at infinity forces
the constant in \eqref{eq:stieltjes-class} to vanish. For the resulting
nonzero representing measure $\nu$, monotone convergence would give
\[
s^\beta K_{a,n}(s)
=\int\left(\frac{s}{s+y}\right)^\beta\nu(dy)
\longrightarrow\nu([0,\infty))\in(0,\infty].
\]
Equation \eqref{eq:leading-asymptotic} instead makes the left-hand side tend
to zero. This proves \eqref{eq:exact-order}.
\end{proof}

\section{Moments of the representing law}

Define the centered spectral quadratic forms
\begin{equation}\label{eq:taus}
\tau_T=\tr(T^2)-\frac{(\tr T)^2}{n},qquad
\tau_X=\tr(X^2)-\frac{(\tr X)^2}{n}.
\end{equation}

\begin{proposition}[Mean and variance]
\label{prop:moments}
The variable in \eqref{eq:Y} satisfies
\begin{equation}\label{eq:mean}
\E Y=\frac{\tr T+\tr X}{n}
\end{equation}
and
\begin{equation}\label{eq:variance}
\operatorname{Var}Y
=\frac{\tau_T+\tau_X}{n(\lambda+1)}.
\end{equation}
\end{proposition}

\begin{proof}
For $n=1$, $V=(1)$ and the variance formula is immediate. Assume $n\ge2$.
Haar averaging gives $\E_U[UVU^*]=I/n$, proving
\eqref{eq:mean}. For traceless Hermitian $A_0,B_0$, Schur orthogonality gives
\begin{equation}\label{eq:haar-variance}
\E_U[\tr(A_0UB_0U^*)^2]
=\frac{\tr(A_0^2)\tr(B_0^2)}{n^2-1}.
\end{equation}
For clarity, invariance makes the left-hand side a scalar multiple of
$\tr(A_0^2)$. Summing over an orthonormal basis of the real Hilbert space of
traceless Hermitian matrices shows that the scalar is
$\tr(B_0^2)/(n^2-1)$, which proves \eqref{eq:haar-variance}.
Conditioning on $V$ therefore yields
\begin{equation}\label{eq:conditional-variance}
\operatorname{Var}(\tr(TUVU^*)\mid V)
=\frac{\tau_T(\tr V^2-1/n)}{n^2-1}.
\end{equation}

It remains to calculate the mean purity. For the unconstrained Laguerre
variables $u_i=rv_i$, integrate the derivative of
\[
u_i^2e^{-\sum_k u_k}\Delta(u)^2\prod_k u_k^{a-1}
\]
over the positive orthant and sum over $i$. Boundary terms vanish for every
$a>0$. The interaction terms reduce to
\[
2\sum_{i<j}\frac{u_i^2-u_j^2}{u_i-u_j}
=2(n-1)\sum_i u_i.
\]
Since $r\sim\operatorname{Gamma}(\lambda,1)$, the resulting identity is
\[
\E\sum_i u_i^2=\lambda(a+2n-1).
\]
Independence of $r$ and $V$, together with
$\E r^2=\lambda(\lambda+1)$, gives
\begin{equation}\label{eq:purity}
\E\tr(V^2)=\frac{a+2n-1}{\lambda+1},
\qquad
\E\left(\tr(V^2)-\frac1n\right)
=\frac{n^2-1}{n(\lambda+1)}.
\end{equation}
The two orbital terms are conditionally independent, and their conditional
means are constant. Taking expectations in
\eqref{eq:conditional-variance} and adding proves
\eqref{eq:variance}.
\end{proof}

\section{Bounds and coefficient inequalities}

Let
\begin{equation}\label{eq:LUm}
L=t_1+x_1,qquad M=t_n+x_n,qquad
m=\frac{\tr T+\tr X}{n},
\end{equation}
and define
\begin{equation}\label{eq:H}
H(s)=\frac{K_{a,n}(s;t,x)}{c_{a,n}\Delta(t)\Delta(x)}.
\end{equation}

\begin{corollary}[Support-and-moment bounds]
\label{cor:bounds}
For $n\ge2$,
\begin{align}
(s+m)^{-\lambda}
&\le H(s)\nonumber\\
&\le
\frac{M-m}{M-L}(s+L)^{-\lambda}
+\frac{m-L}{M-L}(s+M)^{-\lambda}.
\label{eq:jensen-chord}
\end{align}
The lower bound has the refinement
\begin{equation}\label{eq:variance-bound}
H(s)\ge
(s+m)^{-\lambda}
+\frac{\lambda(\tau_T+\tau_X)}{2n}(s+M)^{-\lambda-2}.
\end{equation}
\end{corollary}

\begin{proof}
The function $y\mapsto(s+y)^{-\lambda}$ is strictly convex.
Jensen's inequality gives the lower estimate in
\eqref{eq:jensen-chord}; its chord through $L$ and $M$ gives the upper one.
These are optimal among all probability measures with the given support and
mean. Since
\[
\frac{d^2}{dy^2}(s+y)^{-\lambda}
\ge\lambda(\lambda+1)(s+M)^{-\lambda-2}
\]
on the support, the second-order Taylor lower bound at $m$, followed by
\cref{prop:moments}, proves \eqref{eq:variance-bound}.
\end{proof}

Let $d=M-L$ and $Z=(Y-L)/d$. The full support statement implies
$\supp Z=[0,1]$. Put $\eta_k=\E Z^k$, and write
$(\Delta_f\eta)_k=\eta_{k+1}-\eta_k$ for the forward-difference operator.

\begin{corollary}[Strict Hausdorff hierarchy]
\label{cor:hausdorff}
For $s+L>d$,
\begin{equation}\label{eq:hausdorff-expansion}
(s+L)^\lambda H(s)
=\sum_{k=0}^\infty
(-1)^k\frac{(\lambda)_k}{k!}\eta_k
\left(\frac{d}{s+L}\right)^k.
\end{equation}
The coefficient moments satisfy
\begin{equation}\label{eq:finite-difference}
(-1)^r\Delta_f^r\eta_k
=\E[Z^k(1-Z)^r]>0
\qquad(k,r\ge0).
\end{equation}
All Hankel and interval-localizing moment matrices of $(\eta_k)$ are positive
definite.
\end{corollary}

\begin{proof}
Expand $(1+dZ/(s+L))^{-\lambda}$ by the binomial series and average to obtain
\eqref{eq:hausdorff-expansion}. Equation
\eqref{eq:finite-difference} is the standard finite-difference identity for
Hausdorff moments. Strictness and positive definiteness follow from the full
interval support.
\end{proof}

\section{Chebyshev-Markov determinants and Wronskians}

Let $I\subset[0,\infty)$ be compact, and let $\mu$ be a finite positive
measure whose support contains at least $n$ points. Suppose continuous
functions $\phi_1,\ldots,\phi_n$ form a strict oriented Chebyshev system on
$\supp\mu$:
\begin{equation}\label{eq:chebyshev}
\det[\phi_i(y_j)]_{i,j=1}^{n}>0
\quad\text{for }y_1<\cdots<y_n.
\end{equation}
Define the generalized Markov transforms
\begin{equation}\label{eq:markov-functions}
m_i(s;t)=\int_I\phi_i(y)(s+t+y)^{-a}\,\mu(dy).
\end{equation}

\begin{theorem}[Exact order under Andr\'eief composition]
\label{thm:markov}
For $t_1<\cdots<t_n$ with $t_1+\min I\ge0$,
\begin{equation}\label{eq:markov-det}
D(s)=\det[m_i(s;t_j)]_{i,j=1}^{n}
\end{equation}
has exact Stieltjes order $\lambda=n(a+n-1)$. For every fixed
$t+\min I\ge0$, the confluent Wronskian
\begin{equation}\label{eq:wronskian}
W(s)=\det\left[
\frac{\partial_s^{j-1}m_i(s;t)}{(j-1)!}
\right]_{i,j=1}^{n}
\end{equation}
has the same exact order.
\end{theorem}

\begin{proof}
Andr\'eief's identity gives
\begin{equation}\label{eq:markov-andreief}
D(s)=\frac1{n!}\int_{I^n}
\det[\phi_i(y_k)]
\det[(s+t_j+y_k)^{-a}]
\,\mu^{\otimes n}(dy).
\end{equation}
The product of determinants is symmetric and nonnegative; it is positive on
strictly ordered tuples. Apply \cref{thm:main} to the second determinant.
The result is a finite, nonzero positive mixture of order-$\lambda$ Stieltjes
kernels. Furthermore,
\begin{equation}\label{eq:markov-leading}
\lim_{s\to\infty}s^\lambda D(s)
=\frac{c_{a,n}\Delta(t)}{n!}
\int_{I^n}\det[\phi_i(y_k)]\Delta(y)
\,\mu^{\otimes n}(dy)>0.
\end{equation}
The exactness argument in \cref{thm:main} applies.

Divide \eqref{eq:markov-andreief} by $\Delta(t)$ and let all $t_j$ tend to
$t$. Compactness permits passage to the limit, and the confluent alternant
formula gives \eqref{eq:wronskian}. More explicitly, writing
$Y_y=\operatorname{diag}(y_1,\ldots,y_n)$, the orbital representation in
\cref{thm:main} gives
\begin{equation}\label{eq:wronskian-representation}
W(s)=\frac{c_{a,n}}{n!}\int_{I^n}
\det[\phi_i(y_k)]\Delta(y)
\E_{V,Q}\!\left(s+t+\tr(Y_yQVQ^*)\right)^{-\lambda}
\,\mu^{\otimes n}(dy),
\end{equation}
where $Q$ is normalized-Haar unitary and independent of $V$. The weight
$\det[\phi_i(y_k)]\Delta(y)$ is symmetric and nonnegative, and its integral
is finite and strictly positive. Thus \eqref{eq:wronskian-representation} is
a finite, nonzero positive mixture of order-$\lambda$ Stieltjes kernels. Its
leading mass is the positive quantity in \eqref{eq:markov-leading} after
removing $\Delta(t)$, so the exactness argument in \cref{thm:main} applies.
\end{proof}

\section{Independent normalization checks}

\subsection{Large-shift double alternant}

Expanding at infinity and applying the confluent double-alternant formula
gives
\begin{equation}\label{eq:alternate-leading}
\lim_{s\to\infty}
\frac{s^\lambda K_{a,n}(s;t,x)}{\Delta(t)\Delta(x)}
=\det\left[\frac{(a)_{p+q}}{p!q!}\right]_{p,q=0}^{n-1}.
\end{equation}
The Gamma-moment determinant evaluates as
\begin{equation}\label{eq:gamma-hankel}
\det[(a)_{p+q}]_{p,q=0}^{n-1}
=\prod_{k=0}^{n-1}k!(a)_k.
\end{equation}
Equations \eqref{eq:alternate-leading} and \eqref{eq:gamma-hankel} reproduce
$c_{a,n}$ without using HCIZ.

\subsection{The Cauchy case}

When $a=1$, $c_{1,n}=1$, and Cauchy's determinant formula gives
\begin{equation}\label{eq:cauchy-case}
\frac{K_{1,n}(s;t,x)}{\Delta(t)\Delta(x)}
=\prod_{i,j=1}^{n}(s+t_i+x_j)^{-1}.
\end{equation}
The uniform Dirichlet identity on $n^2$ coordinates represents the right-hand
side as an order-$n^2$ generalized Stieltjes transform. This confirms both the
order and normalization and gives an alternative construction of the law of
$Y$ for $a=1$.

\subsection{Double confluence}

Dividing by the two Vandermonde factors and coalescing both spectra yields
\begin{equation}\label{eq:double-confluence}
\lim_{t_i\to t,\,x_i\to x}
\frac{K_{a,n}(s;t,x)}{\Delta(t)\Delta(x)}
=c_{a,n}(s+t+x)^{-\lambda}.
\end{equation}
This is consistent with the Gamma-Laplace Wronskian evaluated by
\citet[Proposition~2.10]{DimitrovXu2019}.

\section{Novelty boundary and limitations}

Total positivity of the kernel, complete monotonicity of Laplace Wronskians,
the HCIZ formula, and the fixed-trace Laguerre density are established theory.
The proposed contribution is the explicit compact orbital law in
\eqref{eq:main-representation}, its minimal generalized Stieltjes order, the
support and two moments, and the persistence of exact order under the
nonconfluent Chebyshev-Markov composition in \cref{thm:markov}.

Several limitations remain. We do not provide a closed density for $Y$ at
general $a$, the proof is tied to the unitary HCIZ formula, and the literature
comparison requires a full-text check of \citet{GrossRichards1989} and
\citet{DetteMunk1996}. The representation may also admit a pre-existing
formulation in the language of hypergeometric functions of two matrix
arguments or orbital convolutions. These are novelty-review risks rather than
gaps in the proof.
